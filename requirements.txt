# Runtime dependencies. Exact pins, for the same reason image tags are pinned (CLAUDE.md § 5):
# a version resolved at install time is a version nobody chose.
#
# Development/test-only packages live in requirements-dev.txt.

# Postgres driver. psycopg 3, not psycopg2 — the migration runner needs to send a whole migration
# file (multiple statements, one round trip) in a single execute(), which psycopg 3 supports for
# parameterless queries.
psycopg[binary]==3.3.4

# APScheduler 3.x, NOT 4.x. The 4.x line is a rewrite with a different API; SQLAlchemyJobStore,
# EVENT_JOB_MISSED, coalesce, and misfire_grace_time — all four of which this project's
# orchestration contract (CLAUDE.md § 4) names explicitly — are 3.x concepts. The upper bound is
# load-bearing, not decoration.
APScheduler==3.11.3

# Required by APScheduler's SQLAlchemyJobStore. The scheduler's persistent store is the only
# thing in this project that uses SQLAlchemy; application queries use psycopg directly.
SQLAlchemy==2.0.51
